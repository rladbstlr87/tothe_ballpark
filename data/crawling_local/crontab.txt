# ===== Termux cron → proot Ubuntu =====
SHELL=/data/data/com.termux/files/usr/bin/bash
PATH=/data/data/com.termux/files/usr/bin:/data/data/com.termux/files/usr/bin/applets:/system/bin
HOME=/data/data/com.termux/files/home

# (안정화) atd가 꺼져있으면 기동
*/15 * * * * proot-distro login ubuntu -- bash -lc 'pgrep -x atd >/dev/null || /usr/sbin/atd'

# (자정+오전 10시) 오늘 경기 기준으로 at 작업 등록
10 0,10 * * * proot-distro login ubuntu -- bash -lc '"'"'python3 /tothe_ballpark/scripts/schedule_checker.py'"'"'

# (자정) 어제 경기 결과 처리 등 after 작업
0 0 * * * proot-distro login ubuntu -- bash -lc '"'"'cd /tothe_ballpark && /bin/bash after_game.sh >> /tothe_ballpark/data/logs/after_game.log 2>&1'"'"'
